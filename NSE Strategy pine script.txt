//@version=6
//───────────────────────────────────────────────────────────────────────────────
//  NIFTY ORB (Options) – 5m  |  Retest + Re-break + Option Alerts (bucketed)
//───────────────────────────────────────────────────────────────────────────────
strategy("NIFTY ORB (Options) – 5m",
         overlay            = true,
         calc_on_every_tick = true,
         initial_capital = 10000000,
         default_qty_type   = strategy.fixed,
         default_qty_value  = 75,
         commission_type = strategy.commission.cash_per_order,
         commission_value = 185,
         pyramiding         = 2)

// ═════════════ Option params (AUTO month-letter code or MANUAL) ══════════════
optQty = input.int(75, "Option qty (per alert)")

ceil_to_step(v, s)  => s == 0 ? v : math.ceil(v/s)*s
floor_to_step(v, s) => s == 0 ? v : math.floor(v/s)*s

var string TZ = "Asia/Kolkata"

// Webhook payloads
build_json_qty(sym, side, qty, opt) =>
    '{"event":"place_order","order_details":{"symbol":"' + sym + '","qty":' + str.tostring(qty) + ',"type":2,"side":"' + side + '","productType":"INTRADAY","limitPrice":0,"stopPrice":0,"validity":"DAY", "optType": "' + opt + '"}}'

// ══════════════════ ORB inputs ═══════════════════════════════════════════════
stopBufferPts  = input.float(2.0,  "Stop-loss buffer (pts)", step = 0.25)
entryBufferPts = input.float(15.0, "Retest buffer into OR (pts)", step = 0.25)
plotRange      = input.bool(true,  "Plot first-5-min range")
retestExpiry   = input.int (6,     "Cancel sequence after n bars", minval = 1)

// Congestion filter controls
proxPts        = input.float(4.0,  "Near-OR proximity (pts) to call it ‘stuck’", step = 0.25)
stuckBars      = input.int  (3,    "Consecutive bars near OR to block entry", minval = 1, maxval = 6)

finalBar = barstate.isconfirmed

// ══════════════════ Session times (IST) ══════════════════════════════════════
startTime      = timestamp(TZ, year, month, dayofmonth, 9, 15)
endTime        = timestamp(TZ, year, month, dayofmonth, 15, 00)
entryStartTime = timestamp(TZ, year, month, dayofmonth, 9, 25)
entryEndTime   = timestamp(TZ, year, month, dayofmonth, 12, 25)
entryWindow    = time >= entryStartTime and time <= entryEndTime

// ══════════════════ Opening ranges (09:15–:20 / :25 / :30) ══════════════════
newDay   = timeframe.change("D")
var float orHigh    = na
var float orLow     = na
var float orHigh10  = na
var float orLow10   = na
var float orHigh15  = na
var float orLow15   = na
var float orOpen15  = na
var float orClose15 = na
var bool  rangeLocked = false

barH = hour(time, TZ)
barM = minute(time, TZ)
inORbuild   = (barH == 9 and barM >= 15 and barM < 20)
inORbuild10 = (barH == 9 and barM >= 15 and barM < 25)
inORbuild15 = (barH == 9 and barM >= 15 and barM < 30)

if newDay
    orHigh    := na
    orLow     := na
    orHigh10  := na
    orLow10   := na
    orHigh15  := na
    orLow15   := na
    orOpen15  := na
    orClose15 := na
    rangeLocked := false

if inORbuild
    orHigh := math.max(nz(orHigh, high), high)
    orLow  := math.min(nz(orLow , low ),  low )
else if not rangeLocked and not na(orHigh) and not na(orLow)
    rangeLocked := true

if inORbuild10
    orHigh10 := math.max(nz(orHigh10, high), high)
    orLow10  := math.min(nz(orLow10 , low ),  low )

if inORbuild15
    orHigh15 := math.max(nz(orHigh15, high), high)
    orLow15  := math.min(nz(orLow15 , low ),  low )

plot(plotRange and rangeLocked ? orHigh : na, "OR High", color=color.green, linewidth=2)
plot(plotRange and rangeLocked ? orLow  : na,  "OR Low",  color=color.red,   linewidth=2)

// ══════════════════ Flags / helpers / filters ════════════════════════════════
var int longState    = 0   // 0-idle | 1-breakout | 2-retest
var int shortState   = 0
var int longBaseBar  = na
var int shortBaseBar = na

// Retest counters
var int  longRetests     = 0
var int  shortRetests    = 0
var bool longRetouching  = false
var bool shortRetouching = false

thickGreen = close > open and (close - open) / math.max(high - low, 1e-6) >= 0.35
thickRed   = open > close and (open - close) / math.max(high - low, 1e-6) >= 0.35

// True when the 2nd or 3rd 5-min candle extends the 1st candle's range — skips flat/sideways opens.
orExpanded    = orHigh10 > orHigh or orLow10 < orLow or orHigh15 > orHigh or orLow15 < orLow

// Breakout (bar-close gated)
defLongBreak  = rangeLocked and finalBar and open + 6 > orHigh and close > orHigh
defShortBreak = rangeLocked and finalBar and open < orLow + 6  and close < orLow
reLongBreak   = defLongBreak
reShortBreak  = defShortBreak

// Volatility & stretch
atrLen      = 14
atrSeries   = ta.atr(atrLen)
atrMedian   = ta.median(atrSeries, 60)
volOK       = atrSeries > atrMedian

orWidth  = orHigh - orLow
stretch  = orWidth / atrSeries

riskReward = stretch > 1.6 ? 0.75 : 3.0
maxLossInr = input.int(8700, "Catastrophe-stop ₹ risk", minval = 1000, step = 500)
lotQty     = 75
maxLossPts = maxLossInr / lotQty
volOKVol   = volume > ta.sma(volume, 20) * 0.3

// ══════════════════ DIRECTION BUCKETS (aggregate per day) ════════════════════
// Long (CE) bucket
var bool   L_active   = false
var string L_sym      = ""
var string L_opt      = ""
var int    L_qty      = 0
var float  L_sl       = na
var float  L_tp       = na
var int    L_openBar  = na  // last entry bar for same-bar guard

// Short (PE) bucket
var bool   S_active   = false
var string S_sym      = ""
var string S_opt      = ""
var int    S_qty      = 0
var float  S_sl       = na
var float  S_tp       = na
var int    S_openBar  = na

// Reset buckets each new day
if newDay
    L_active := false, L_sym := "", L_opt := "", L_qty := 0, L_sl := na, L_tp := na, L_openBar := na
    S_active := false, S_sym := "", S_opt := "", S_qty := 0, S_sl := na, S_tp := na, S_openBar := na
    longState  := 0
    shortState := 0
    longRetests := 0
    shortRetests := 0
    longRetouching := false
    shortRetouching := false

// ══════════════════ Stuck-bars (keep) ════════════════════════════════════════
near5Hi   = not na(orHigh)   and math.abs(low  - orHigh)  <= proxPts
near15Hi  = not na(orHigh15) and math.abs(low  - orHigh15)<= proxPts
near5Lo   = not na(orLow)    and math.abs(high - orLow)   <= proxPts
near15Lo  = not na(orLow15)  and math.abs(high - orLow15) <= proxPts

nearRes = (near5Hi or near15Hi) ? 1.0 : 0.0
nearSup = (near5Lo or near15Lo) ? 1.0 : 0.0

stuckLong  = nz(ta.sma(nearRes, stuckBars) * stuckBars, 0) == stuckBars
stuckShort = nz(ta.sma(nearSup, stuckBars) * stuckBars, 0) == stuckBars

plotshape(stuckLong,  "StuckLong",  shape.circle, color=color.orange, location=location.abovebar, size=size.tiny)
plotshape(stuckShort, "StuckShort", shape.circle, color=color.aqua,   location=location.belowbar, size=size.tiny)


// ══════════════════ Retest counting (bar-close) ══════════════════════════════
if finalBar
    if (longState == 1 or longState == 2)
        longTouchNow = (low <= orHigh + entryBufferPts)
        if longTouchNow and not longRetouching
            longRetests += 1
            longRetouching := true
        else if not longTouchNow
            longRetouching := false
    if (shortState == 1 or shortState == 2)
        shortTouchNow = (high >= orLow - entryBufferPts)
        if shortTouchNow and not shortRetouching
            shortRetests += 1
            shortRetouching := true
        else if not shortTouchNow
            shortRetouching := false

// ══════════════════ Main logic (entries bar-close confirmed) ═════════════════
if rangeLocked and orExpanded
    // LONG
    if longState == 0 and defLongBreak and volOK
        longState      := 1
        longBaseBar    := bar_index
        longRetests    := 0
        longRetouching := false
    else if longState == 1
        if bar_index - longBaseBar > retestExpiry
            longState := 0
            longRetouching := false
            longRetests := 0
        if finalBar and low <= orHigh + entryBufferPts
            longState := 2
    else if longState == 2
        if bar_index - longBaseBar > retestExpiry
            longState := 0
            longRetouching := false
            longRetests := 0
        if finalBar and reLongBreak and thickGreen and high[1] > high[2] and volOKVol and entryWindow and (longRetests <= 2) and not stuckLong
            entry  = close
            slBase = (orLow + stopBufferPts)
            risk   = entry - slBase
            tp     = entry + risk * riskReward
            catStopLong = entry - maxLossPts
            actSL_long  = catStopLong > slBase ? catStopLong : slBase

            // Strategy position (unchanged)
            strategy.entry("Long", strategy.long,
                           comment = "d=" + str.tostring(bar_index - longBaseBar) + ",r=" + str.tostring(longRetests) + ",s=" + str.tostring(stretch))
            strategy.exit("L-Exit", from_entry="Long", stop = actSL_long, limit = tp, comment = "TP/SL")

            // ── Bucketed alerts (aggregate qty, single symbol per day) ──
            if not L_active
                // First long of the day → pick symbol and seed bucket
                L_sym     := "NSE:NIFTY50-INDEX"
                L_opt     := "CE"
                L_qty     := optQty
                L_sl      := actSL_long
                L_tp      := tp
                L_openBar := bar_index
                L_active  := true
                alert(build_json_qty(L_sym, "1", optQty, L_opt), alert.freq_once_per_bar)
            else
                // Add-on: use same symbol, accumulate qty, tighten stops if needed
                L_sym     := "NSE:NIFTY50-INDEX"
                L_opt     := "CE"
                L_qty += optQty
                L_sl   := math.max(L_sl, actSL_long) // for longs, tighter (higher) stop dominates
                // Keep TP as the *nearest* of existing/new to increase chance of a single exit
                L_tp   := math.min(L_tp, tp)
                L_openBar := bar_index
                alert(build_json_qty(L_sym, "1", optQty, L_opt), alert.freq_once_per_bar)

            longState := 0
            longRetouching := false
            longRetests := 0

    // SHORT
    if shortState == 0 and defShortBreak and volOK
        shortState      := 1
        shortBaseBar    := bar_index
        shortRetests    := 0
        shortRetouching := false
    else if shortState == 1
        if bar_index - shortBaseBar > retestExpiry
            shortState := 0
            shortRetouching := false
            shortRetests := 0
        if finalBar and high >= orLow - entryBufferPts
            shortState := 2
    else if shortState == 2
        if bar_index - shortBaseBar > retestExpiry
            shortState := 0
            shortRetouching := false
            shortRetests := 0
        if finalBar and reShortBreak and thickRed and low[1] < low[2] and volOKVol and entryWindow and (shortRetests <= 2) and not stuckShort
            entry  = close
            slBase = (orHigh - stopBufferPts)
            risk   = slBase - entry
            tp     = entry - risk * riskReward
            catStopShort = entry + maxLossPts
            actSL_short  = catStopShort < slBase ? catStopShort : slBase

            strategy.entry("Short", strategy.short,
                           comment = "d=" + str.tostring(bar_index - shortBaseBar) + ",r=" + str.tostring(shortRetests) + ",s=" + str.tostring(stretch))
            strategy.exit("S-Exit", from_entry="Short", stop = actSL_short, limit = tp, comment = "TP/SL")

            // Bucketed alerts for shorts
            if not S_active
                S_sym     := "NSE:NIFTY50-INDEX"
                S_opt     := "PE"
                S_qty     := optQty
                S_sl      := actSL_short
                S_tp      := tp
                S_openBar := bar_index
                S_active  := true
                alert(build_json_qty(S_sym, "1", optQty, S_opt), alert.freq_once_per_bar)
            else
                S_sym     := "NSE:NIFTY50-INDEX"
                S_opt     := "PE"
                S_qty += optQty
                S_sl   := math.min(S_sl, actSL_short) // for shorts, tighter (lower) stop dominates
                S_tp   := math.max(S_tp, tp)          // nearest TP for shorts is the larger (less room down)
                S_openBar := bar_index
                alert(build_json_qty(S_sym, "1", optQty, S_opt), alert.freq_once_per_bar)

            shortState := 0
            shortRetouching := false
            shortRetests := 0

// ══════════════════ Intrabar exit monitor (bucket-level, single alert) ═══════
// Guard against same-bar entry→exit by comparing to last entry bar.

if L_active
    if bar_index != L_openBar
        // Long: exit if TP hit (high >= L_tp) or SL hit (low <= L_sl)
        if (high >= L_tp) or (low <= L_sl)
            alert(build_json_qty(L_sym, "-1", L_qty, L_opt), alert.freq_once_per_bar)
            L_active := false, L_sym := "", L_opt := "", L_qty := 0, L_sl := na, L_tp := na, L_openBar := na

if S_active
    if bar_index != S_openBar
        // Short: exit if TP hit (low <= S_tp) or SL hit (high >= S_sl)
        if (low <= S_tp) or (high >= S_sl)
            alert(build_json_qty(S_sym, "-1", S_qty, S_opt), alert.freq_once_per_bar)
            S_active := false, S_sym := "", S_opt := "", S_qty := 0, S_sl := na, S_tp := na, S_openBar := na


// ════════════════ Combined max loss across all open trades ══════════════════
// If total OPEN loss (sum of all negative P/L) reaches -maxLossInr, flatten.

if strategy.opentrades > 0
    float totalOpenLoss = 0.0

    // Sum only the losing trades
    for tradeNo = 0 to strategy.opentrades - 1
        thisPl = strategy.opentrades.profit(tradeNo)  // P/L in INR
        if thisPl < 0
            totalOpenLoss += thisPl  // thisPl is negative, so totalOpenLoss is negative

    // Example: trade1 = -4000, trade2 = -5400 → totalOpenLoss = -9400
    if totalOpenLoss <= -maxLossInr
        // Flatten all strategy positions
        strategy.close_all(comment = "Combined max loss hit")

        // Flatten options buckets + send exit alerts

        if L_active
            alert(build_json_qty(L_sym, "-1", L_qty, L_opt), alert.freq_once_per_bar)
            L_active  := false
            L_sym     := ""
            L_opt     := ""
            L_qty     := 0
            L_sl      := na
            L_tp      := na
            L_openBar := na

        if S_active
            alert(build_json_qty(S_sym, "-1", S_qty, S_opt), alert.freq_once_per_bar)
            S_active  := false
            S_sym     := ""
            S_opt     := ""
            S_qty     := 0
            S_sl      := na
            S_tp      := na
            S_openBar := na



// ══════════════════ EOD one-shot: close outstanding buckets ═════════════════
var bool eodAlertSent = false
if newDay
    eodAlertSent := false

if time >= endTime and not eodAlertSent
    if L_active
        alert(build_json_qty(L_sym, "-1", L_qty, L_opt), alert.freq_once_per_bar)
        L_active := false, L_sym := "", L_opt := "", L_qty := 0, L_sl := na, L_tp := na, L_openBar := na
    if S_active
        alert(build_json_qty(S_sym, "-1", S_qty, S_opt), alert.freq_once_per_bar)
        S_active := false, S_sym := "", S_opt := "", S_qty := 0, S_sl := na, S_tp := na, S_openBar := na
    eodAlertSent := true

// ══════════════════ Strategy forced flat at session end ══════════════════════
if time >= endTime
    strategy.close_all(comment = "w = " + str.tostring(orWidth))
